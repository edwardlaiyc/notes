% Cheatsheet profile for /md-to-latex — landscape four-column reference card.
% Filled blue section bars (white text), bold run-in subsections, boxed blocks,
% tight lists. No title, no TOC, no page numbers. Only placeholder: <<INPUTS>>.
% Cross-compiler safe (pdflatex + tectonic/xelatex).
\documentclass[8pt]{extarticle}

\usepackage{iftex}
\ifpdftex
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
\fi

\usepackage[a4paper,landscape,margin=0.5cm]{geometry}
\usepackage{multicol}
\usepackage{amsmath, amssymb}
\usepackage{graphicx}
\usepackage{xcolor}
\usepackage[most]{tcolorbox}
\usepackage{enumitem}
\usepackage[explicit]{titlesec}

% Four columns with a faint separating rule.
\setlength{\columnsep}{0.7em}
\setlength{\columnseprule}{0.2pt}

% Filled blue section bar spanning the column: white bold text on solid blue.
\definecolor{secblue}{RGB}{0,0,205}
\titleformat{\section}
  {\normalfont\footnotesize\bfseries\color{white}}{}{0pt}
  {\colorbox{secblue}{\parbox{\dimexpr\columnwidth-2\fboxsep\relax}{\raggedright #1}}}
\titlespacing*{\section}{0pt}{4pt}{2pt}

% Subsections: bold run-in labels, no color.
\titleformat{\subsection}[runin]{\normalfont\bfseries}{}{0pt}{#1}[]
\titlespacing*{\subsection}{0pt}{2pt}{0.5em}

% Tight lists.
\setlist[itemize]{leftmargin=1.1em, topsep=1pt, itemsep=0pt, parsep=0pt}
\setlist[enumerate]{leftmargin=1.4em, topsep=1pt, itemsep=0pt, parsep=0pt}

% Boxed theorem-like block: markdown "> ..." maps here (delta).
\newtcolorbox{mdbox}{colback=white, colframe=black, boxrule=0.4pt,
  left=3pt, right=3pt, top=2pt, bottom=2pt, arc=1pt, before skip=2pt, after skip=2pt}

% Centered in-place image (kept for parity with the engine's image rule).
\newcommand{\mdfigure}[3][\linewidth]{\begin{center}\includegraphics[width=#1]{#2}\\#3\end{center}}
\newcommand{\mdfigureplain}[2][\linewidth]{\begin{center}\includegraphics[width=#1]{#2}\end{center}}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{2pt}

\begin{document}
\begin{multicols}{4}
<<INPUTS>>
\end{multicols}
\end{document}
