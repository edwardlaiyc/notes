% Plain-article profile for /md-to-latex — clean single-column notes.
% Article class, Computer Modern, numbered sections, \maketitle,
% centered in-place images with a plain caption below.
% Defines the \mdfigure/\mdfigureplain macros the engine's fragments call.
% Cross-compiler safe: compiles under pdflatex AND tectonic/xelatex.
\documentclass[11pt]{article}

\usepackage{iftex}
\ifpdftex
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
\fi

\usepackage{amsmath, amssymb}   % math passthrough from $...$ and $$...$$
\usepackage{graphicx}           % \includegraphics
\usepackage{enumitem}           % list spacing
\usepackage{hyperref}           % [text](url) -> \href

% Centered, in-place image with a plain caption below (no float, no "Figure N:").
% Optional first arg overrides the width.
\newcommand{\mdfigure}[3][0.8\textwidth]{%
  \begin{center}\includegraphics[width=#1]{#2}\\[4pt]#3\end{center}}

% Same, but no caption line (for images with empty markdown alt text).
\newcommand{\mdfigureplain}[2][0.8\textwidth]{%
  \begin{center}\includegraphics[width=#1]{#2}\end{center}}

\title{<<TITLE>>}
\date{}
\author{}

\begin{document}
\maketitle
<<TOC>>
<<INPUTS>>
\end{document}
