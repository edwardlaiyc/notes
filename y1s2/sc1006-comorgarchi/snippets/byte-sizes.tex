\begin{enumerate}
    \item 1 byte = 8 bits
    \item 1KB = $2^{10}$ bytes = 1024 bytes
    \item 1MB = $2^{20}$ bytes = 1,048,576 bytes
    \item 1GB = $2^{30}$ bytes = 1,073,741,824 bytes
\end{enumerate}
