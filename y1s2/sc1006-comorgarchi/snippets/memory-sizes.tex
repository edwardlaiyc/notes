\begin{itemize}
    \item Size at each address = 8 bits
    \item Size of address (\textit{assumed}) = 16 bits
    \[
    \text{Size of pointer}_{\text{bytes}} = 8 \times \left\lceil \frac{\text{Address Bits}}{8} \right\rceil
    \]
    \item Size of each register in CPU = 32 bits.\newline
    There are 16 registers in CPU (R0-R12; R13=SP; R14=LP; R15=PC; CSPR)
    \item Memory capacity in BYTES = $2^{\text{Number of address pins}}$
    \item Number of BITS in one data transfer = number of data pins
\end{itemize}
