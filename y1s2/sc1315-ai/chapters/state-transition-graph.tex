class VacuumWorldState:
    def __init__(self, 
                 dirtLeft=True, 
                 dirtRight=True, 
                 robotLeft=True):
        self.dirtLeft = dirtLeft
        self.dirtRight = dirtRight
        self.robotLeft = robotLeft

    def robotRight(self):
        return not self.robotLeft
