\section{Random Variables and Distributions}

\subsection*{Random variables}
A random variable is a function that maps outcomes from a sample space to a real number.
\[X: \Omega \rightarrow \mathbb{R}\]

\subsubsection*{Example}
\noindent Rolling two dice, we have $\Omega = {(i, j)\mid i,j\in \{1,2,3,4,5,6\}}$.

\noindent We can define several random variables on $\Omega$.
\begin{enumerate}
    \item Sum: $X((i,j)) = i + j$
    \item Larger of the two: $X((i, j)) = \max(i, j)$
    \item Indicator of doubles: $X(\omega) = \begin{cases}
        1 & \text{if }i = j \\
        0 & \text{else}
    \end{cases}$
\end{enumerate}

\bigskip

\paragraph{Discrete Random Variables} X has a finite or countably infinite range of possible values.
\[\text{Range}(X) = \{n_1, n_2, \ldots, n_k\} \text{ or } \{n_1, n_2, \ldots\}\]

\paragraph{Continuous Random Variables} X takes values in an uncountable subset/interval of $\mathbb{R}$.
\[\text{Range}(X) \subset \mathbb{R}\text{, an interval/continuum}\]

\subsubsection*{Probability law of random variables}
For a probability space $(\Omega, P)$, and random variable $X: \Omega \rightarrow \mathbb{R}$,

\noindent and for any set $A \subseteq \mathbb{R}$, the probability law/distribution of X is

\[P_X(A) = P(X\in A) = P(X^{-1}(A)) = P(\{\omega\in\Omega: X(\omega)\in A\})\]

\noindent Instead of discussing $P$ on $\Omega$, we look at the push-forward measure
of $P$ by $X$, where $P_X(A)$ is the likelihood of event $A\subseteq\mathbb{R}$ occuring.

\noindent The corresponding sample space is $\Omega' = \text{Range}(X) = \{X(\omega)\mid \omega\in\Omega\}$.

\noindent We say that $X$ follows the probability distribution $P_X$.

\medskip

\noindent Proof that $P_X$ is a valid probability measure:
\begin{enumerate}
    \item $0\leq P_X(A)\leq 1$
    \item $P_X(\Omega') = P(X^{-1}(\Omega')) = P(\Omega) = 1$
    \item For $A_1, A_2,\ldots\subset\Omega',\: A_i\cap A_j=\emptyset,\:i\neq j$, (disjoint sets)\newline 
    $P_X(\cup A_i) = P(X^{-1}(\cup A_{i})) = P(\cup X^{-1}(A_i))
    = \sum P(X^{-1}(A_i)) = \sum P_X(A_i)$
\end{enumerate}

\subsection{Cumulative Distribution Function}
The CDF of some discrete/continuous random variable $X$ is
\[F_X(t) = P_X(\{x\mid x\leq t\}) = P(X\leq t) = P(X\in (-\infty,t])\]

\noindent Properties:
\begin{enumerate}
    \item $F_X(t)$ is non-decreasing.
    \item $\lim\limits_{t\to -\infty}F_X(t) = 0,\: \lim\limits_{t\to +\infty}F_X(t) = 1$
    \item $F_X(t)$ is right-continuous ($\lim\limits_{x\to t^{+}}F(x) = F(t)$).
\end{enumerate}
$P(X\in (a, b]) = F_X(b) - F_X(a)$.

\subsection{Probability Mass Function}
The PMF of some discrete random variable $X$ is
\[p_X(x) = P_X({x}) = P(X = x)\]

    \centering

    \caption*{Probability Mass Function} % \caption* removes "Figure 1:" prefix
\end{figure}

\noindent Properties:
\begin{enumerate}
    \item $p_X(x) \geq 0$
    \item $\sum\limits_{x\in \text{Range}(X)}p_X(x) = 1$
    \item $F_X(x) = \sum\limits_{t\leq x}p_X(t)$ \newline
    E.g. For the probability that a dice roll is 3 or less,
    we sum the PMF values for 1, 2, 3.
\end{enumerate}

\noindent Referring to the discrete CDF graph, the height of each vertical
jump is equal to the probability $p_X(x)$ of that specific value.

\smallskip

\noindent The CDF $F_X(x)$ is right-continuous to ensure that 
$F_X(x) = P(X \leq x)$ includes the probability of $x$ itself 
($P(X=x)$).

\medskip

\noindent Example for number of "heads" in 2 coin flips:

$\Omega = \{(H,H), (H,T), (T,H), (T,T)\}$, 

$X(\omega)$ = Number of $H$

$\Omega' = \text{Range}(X) = \{0,1,2\}$
\[p_X(0) = \frac{1}{4},\:p_X(1) = \frac{1}{2}, \:p_X(2) = \frac{1}{4}\]


\subsubsection{Bernoulli Distribution}
A discrete random variable $X$ with a Bernoulli Distribution takes two values,
\[X\in \{0,1\}\]
and with probability mass function
\[P(X=1) = p,\:P(X=0)=1-p\]
\noindent Used to model scenarios with binary outcomes, like Heads or Tails.

\subsubsection{Categorical Distribution}
A random variable $X$ with a Categorical Distribution takes one of K-discrete categories,
\[X\in \{1,2,\ldots,K\}\]
and with probability mass function
\[P(X=i) = p_i,\hspace{0.5cm}p_i\geq 0,\hspace{0.5cm} \sum_{i=1}^K p_i=1\]
\noindent It is a multi-class extension of the Bernoulli Distribution. When $K=2$,
it becomes the Bernoulli Distribution.

\subsection{Probability Density Function}
The PDF of some continuous random variable $X$ is, (given some event A)
\[f_X(A) = P(X\in A) = \int_A f_X(x)\,\mathrm{d}x\]

\noindent Properties:
\begin{enumerate}
    \item $f_X(x) = \frac{\mathrm{d}}{\mathrm{d}x}F_X(x)$, when $f_X$ is continuous. \newline
    PDF graph is the derivative of CDF graph, and CDF is the integral of the PDF graph.
    \item $f_X(x) \geq 0$
    \item $\int_{-\infty}^{+\infty}f_X(x)\mathrm{d}x = 1$
    \item $F_X(x) = P_X((-\infty, x]) = \int_{-\infty}^{x} f_X(t) \, dt$
\end{enumerate}

\subsection{Gaussian/Normal Distribution}
The Gaussian Distribution has a symmetric bell curve shape.

\noindent Since it is a continous probability distribution, it can be modelled by a PDF or CDF.

\noindent Gaussian/normal distribution $N(\mu,\sigma^2)$
\[ f_X(x) = \frac{1}{\sqrt{2\pi\sigma^2}} \exp(-\frac{(x-\mu)^2}{2\sigma^2}) \]

\begin{itemize}
    \item $\mu$ is the mean (center of the distribution / long-run arithmetic average value).
    \item $\sigma^2$ is the variance (spread of the distribution).
    \item The term $\frac{1}{\sqrt{2\pi\sigma^2}}$ is the normalization constant required to satisfy the condition $\int p(x)dx = 1$.
\end{itemize}

\noindent The \textbf{Standard Gaussian distribution} is $N(0,1)$.
\[f_{X}(x)=\frac{1}{\sqrt{2\pi}}\exp(\frac{-x^2}{2})\]

\paragraph{Proportionality ($\propto$)}
The constant normalization factors is usually omitted for simplification. A PDF is often written as
\[
p(x) = \frac{g(x)}{Z} \Rightarrow p(x) \propto g(x)
\]
\begin{itemize}
    \item $g(x)$ is the \textbf{unnormalized function} (containing $x$).
    \item $Z$ is the \textbf{normalization constant} (independent of $x$), defined as $Z = \int g(x)dx$.
\end{itemize}

\noindent By extension, we can identify a Gaussian by Inspection.
If a function has the quadratic form in its exponent, it is a Gaussian distribution, even if the constant is missing. \newline
Example:
\[
p(x) \propto \exp\left( -\frac{1}{2\sigma^2}x^2 + \frac{\mu}{\sigma^2}x \right)
\]
By completing the square in the exponent, we can identify this as $\mathcal{N}(\mu, \sigma^2)$:
\[
\text{RHS} = \exp\left( -\frac{1}{2\sigma^2}(x - \mu)^2 \right) \cdot \exp\left( \frac{\mu^2}{2\sigma^2} \right)
\]
The term $\exp\left( \frac{\mu^2}{2\sigma^2} \right)$ is constant with respect to $x$ and can be absorbed into the proportionality constant.

\subsubsection{Gaussian Integral}
To prove that the Gaussian PDF integrates to 1, we use the standard Gaussian integral result:
\[
\int_{-\infty}^{\infty} e^{-x^2} \, dx = \sqrt{\pi}
\]
By applying the change of variable $y = \frac{1}{\sqrt{2\sigma^2}}(x - \mu)$, we can derive:
\[
\int_{-\infty}^{\infty} \exp\left( -\frac{1}{2\sigma^2}(x - \mu)^2 \right) \, dx = \sqrt{2\pi\sigma^2}
\]
The general form of the Gaussian integral is
\[
\int_{-\infty}^{\infty} e^{-ax^2} \, dx = \sqrt{\frac{\pi}{a}}
\]

\subsection{Bayesian Inference}
\subsubsection*{Example}
\begin{itemize}
    \item \textbf{Context:} A factory produces products labeled with a weight of 100g.
    \item \textbf{Data (Observations):} We sample $n=3$ products with weights:
    $$ x_1 = 99\text{g}, \quad x_2 = 101\text{g}, \quad x_3 = 103\text{g} $$
    \item \textbf{Likelihood Model:} The actual weight $X$ is known to follow a Gaussian distribution with variance $\sigma^2 = 1$.
    $$ X \sim \mathcal{N}(\mu, 1) $$
    \item \textbf{Prior Belief:} Before seeing the data, we model our uncertainty about the mean $\mu$ using a Prior distribution.
    $$ \mu \sim \mathcal{N}(\mu_0, \sigma_0^2) = \mathcal{N}(100, 4) $$
    i.e. we are guessing that the mean is 100 with a variance of 4.
\end{itemize}

\subsubsection{The Goal}
We wish to find the \textbf{posterior distribution} $P(\mu \mid x_1, \dots, x_n)$, which represents our updated belief about the mean $\mu$ after observing the data.

\subsubsection*{Bayes' Rule Formulation}
Using Bayes' Rule, the posterior is proportional to the likelihood times the prior:
\[
P(\mu \mid x_1, \dots, x_n) = \frac{P(x_1, \dots, x_n \mid \mu) P(\mu)}{P(x_1, \dots, x_n)}
\]
Since the denominator $P(x_1, \dots, x_n)$ does not depend on $\mu$, we treat it as a normalization constant and write:
\[
P(\mu \mid x_1, \dots, x_n) \propto P(x_1, \dots, x_n \mid \mu) \cdot P(\mu)
\]

\subsubsection*{Defining the Components}

1. The Likelihood Function \\
Since the samples are independent, the joint probability is the product of individual probabilities:
\begin{align*}
P(x_1, \dots, x_n \mid \mu) &= \prod_{i=1}^{n} P(x_i \mid \mu) \\
&= \prod_{i=1}^{n} \frac{1}{\sqrt{2\pi\sigma^2}} \exp\left( -\frac{1}{2\sigma^2}(x_i - \mu)^2 \right) \\
&\propto \exp\left( -\frac{1}{2\sigma^2} \sum_{i=1}^{n} (x_i - \mu)^2 \right)
\end{align*}

\noindent 2. The Prior Function \\
Our prior belief is a single Gaussian centered at $\mu_0$:
\begin{align*}
P(\mu) &= \frac{1}{\sqrt{2\pi\sigma_0^2}} \exp\left( -\frac{1}{2\sigma_0^2}(\mu - \mu_0)^2 \right) \\
&\propto \exp\left( -\frac{1}{2\sigma_0^2}(\mu - \mu_0)^2 \right)
\end{align*}

\subsubsection*{The Combined Equation}
Multiplying the Likelihood and Prior (by adding their exponents):
\[
P(\mu | \text{data}) \propto \exp\left( \left[ -\frac{1}{2\sigma^2} \sum_{i=1}^{n} (x_i - \mu)^2 \right] + \left[ -\frac{1}{2\sigma_0^2}(\mu - \mu_0)^2 \right] \right)
\]

\subsubsection{The math}
\subsubsection*{Expanding the Squares}
We expand the quadratic terms. Note that any term \textbf{not} containing $\mu$ is effectively a constant relative to $\mu$ and can be absorbed into the proportionality constant.
\begin{align*}
\sum_{i=1}^{n}(x_i - \mu)^2 &= \sum x_i^2 - 2\mu\sum x_i + n\mu^2 \\
(\mu - \mu_0)^2 &= \mu^2 - 2\mu\mu_0 + \mu_0^2
\end{align*}
Discarding terms without $\mu$ (like $\sum x_i^2$ and $\mu_0^2$), the relevant part of the exponent is:
\[
-\frac{1}{2} \left[ \frac{n\mu^2 - 2\mu\sum x_i}{\sigma^2} + \frac{\mu^2 - 2\mu\mu_0}{\sigma_0^2} \right]
\]

\subsubsection*{Grouping Coefficients}
We group the terms by $\mu^2$ and $\mu$:
\[
-\frac{1}{2} \left[ \mu^2 \left( \frac{n}{\sigma^2} + \frac{1}{\sigma_0^2} \right) - 2\mu \left( \frac{\sum x_i}{\sigma^2} + \frac{\mu_0}{\sigma_0^2} \right) \right]
\]
This matches the standard Gaussian form $-\frac{1}{2\sigma_{new}^2}(\mu - \mu_{new})^2$, which expands to $-\frac{1}{2\sigma_{new}^2}(\mu^2 - 2\mu\mu_{new})$.

\subsubsection{The posterior parameters}
By comparing coefficients, we derive the Posterior Variance ($\sigma_n^2$) and Mean ($\mu_n$):

\noindent\textbf{Posterior Variance ($\sigma_n^2$):}
\[
\frac{1}{\sigma_n^2} = \frac{n}{\sigma^2} + \frac{1}{\sigma_0^2} \quad \Rightarrow \quad \sigma_n^2 = \frac{\sigma^2 \sigma_0^2}{n\sigma_0^2 + \sigma^2}
\]

\noindent\textbf{Posterior Mean ($\mu_n$):}
\[
\frac{\mu_n}{\sigma_n^2} = \frac{\sum x_i}{\sigma^2} + \frac{\mu_0}{\sigma_0^2} \quad \Rightarrow \quad \mu_n = \sigma_n^2 \left( \frac{\sum x_i}{\sigma^2} + \frac{\mu_0}{\sigma_0^2} \right)
\]

\subsubsection*{Numerical Calculation}
We apply the values from the problem:
\begin{itemize}
    \item $n = 3$, $\sum x_i = 303$
    \item Data: $\sigma^2 = 1$
    \item Prior: $\mu_0 = 100$, $\sigma_0^2 = 4$
\end{itemize}

\noindent\textbf{Calculate Variance:}
\begin{equation*}
\sigma_n^2 = \frac{1 \cdot 4}{3(4) + 1} = \frac{4}{13} \approx 0.308
\end{equation*}

\noindent\textbf{Calculate Mean:}
\begin{equation*}
\mu_n = \frac{4}{13} \left( \frac{303}{1} + \frac{100}{4} \right) = \frac{4}{13} (303 + 25) = \frac{4}{13}(328) = \frac{1312}{13} \approx 100.92
\end{equation*}

\noindent\textbf{Conclusion:}
The posterior distribution is $\mu \sim \mathcal{N}(100.92, 0.308)$.

\subsubsection{Interpretationss}
The formulas:
\begin{align*}
\text{Variance:} \quad \hat{\sigma}_n^2 &= \frac{\sigma^2 \sigma_0^2}{n\sigma_0^2 + \sigma^2} \\
\text{Mean:} \quad \hat{\mu}_n &= \hat{\sigma}_n^2 \left( \frac{\sum x_i}{\sigma^2} + \frac{\mu_0}{\sigma_0^2} \right)
\end{align*}

\subsubsection*{1. Prior unconfidence ($\sigma_0^2 \to \infty$)}
Imagine we have zero confidence in our prior belief (variance is infinite).
\begin{itemize}
    \item As $\sigma_0^2 \to \infty$, the term $\frac{1}{\sigma_0^2} \to 0$.
    \item The prior effectively vanishes from the equation.
\end{itemize}

\noindent\textbf{Limit of the Variance:}
\begin{equation*}
\lim_{\sigma_0^2 \to \infty} \hat{\sigma}_n^2 = \frac{\sigma^2}{n}
\end{equation*}

\noindent\textbf{Limit of the Mean:}
\begin{equation*}
\lim_{\sigma_0^2 \to \infty} \hat{\mu}_n = \frac{\sigma^2}{n} \left( \frac{\sum x_i}{\sigma^2} + 0 \right) = \frac{1}{n} \sum_{i=1}^{n} x_i = \bar{x}
\end{equation*}

\noindent\textbf{Interpretation:}
This recovers the \textbf{Frequentist Result}. Without a prior, our best guess is simply the sample mean ($\bar{x}$) and the standard error depends purely on the data variance.

\subsubsection*{2. Infinite Data ($n \to \infty$)}
Imagine we collect a massive amount of data.
\begin{itemize}
    \item The prior information ($\mu_0, \sigma_0^2$) becomes negligible compared to the weight of the evidence.
    \item We are more confident that the real mean is the sample mean.
    \item Note that actual data (i.e. the factory machine) still has its own variance. Our variance is the representation of our confidence.
\end{itemize}

\noindent\textbf{Limit of the Variance:}
\begin{equation*}
\lim_{n \to \infty} \hat{\sigma}_n^2 = \lim_{n \to \infty} \frac{\sigma^2 \sigma_0^2}{n\sigma_0^2 + \sigma^2} = 0
\end{equation*}
(Our uncertainty shrinks to zero; we become perfectly confident).

\noindent\textbf{Limit of the Mean:}
\begin{align*}
\lim_{n \to \infty} \hat{\mu}_n &= \lim_{n \to \infty} \hat{\sigma}_n^2 \left( \frac{\sum_{i=1}^n x_i}{\sigma^2} + \frac{\mu_0}{\sigma_0^2} \right) \\
&= \lim_{n \to \infty} \left( \frac{\sigma^2 \sigma_0^2}{n\sigma_0^2 + \sigma^2} \right) \frac{n\bar{x}}{\sigma^2} + \left( \frac{\sigma^2 \sigma_0^2}{n\sigma_0^2 + \sigma^2} \right) \frac{\mu_0}{\sigma_0^2} \\
&= \lim_{n \to \infty} \frac{\sigma_0^2 \bar{x}}{\sigma_0^2 + \frac{\sigma^2}{n}} + 0 \\
&= \bar{x}
\end{align*}
(The estimate converges to the true empirical average).

\subsubsection{The core philosophy of Bayesian Inference}
The prior distribution compensates for the lack of data when the sample size is small.
\begin{itemize}
    \item \textbf{Small $n$:} The Prior $\mu_0$ pulls the estimate towards our initial belief.
    \item \textbf{Large $n$:} Large sample size dominates, and the Prior is eventually forgotten.
\end{itemize}

\subsection{Multivariate Random Variables}
We now consider two random variables, $X$ and $Y$, and assign probability to $Z=(X,Y)$.

\noindent For any set $A\in\mathbb{R}^2$,
\[P_{(X,Y)}(A) = P((X,Y)\in A) = P((X,Y)^{-1}A) = P(\{\omega\in\Omega:(X(\omega),Y(\omega))\in A\})\]
This is a joint distribution on $X,Y$.

\medskip

\subsubsection{Cumulative Distributive Function} For two random variables $X,Y$, their CDF is
\[F_{X,Y}(x,y) = P(X\leq x,Y\leq y) = P(X\in (-\infty,x],Y\in(-\infty,y])\]
\[F_{X,Y}: \mathbb{R}^2 \rightarrow \mathbb{R}\]

\subsubsection{Joint Distribution/PMF (Discrete)} For two random variables $X,Y$, their joint PMF is
\[p_{X,Y}(x,y) = P_{X,Y}(\{x,y\}) = P(X=x, Y=y)\]

\noindent Properties:
\begin{enumerate}
    \item $p_{X,Y}(x,y) \geq 0$
    \item $\sum\limits_{(x,y)\in\text{Range}(X,Y)}p_{X,Y}(x,y) = 1$
\end{enumerate}

\subsubsection{Marginal Distribution/PMF (Discrete)}
Given the joint PMF $p_{X,Y}(x,y)$, the marginal distributions/PMFs of $X$ and $Y$
are found by summing over all possible values of the other variable.
\[p_X(x) = \sum\limits_{y}p_{X,Y}(x,y)\]
\[p_Y(y) = \sum\limits_{x}p_{X,Y}(x,y)\]

\subsubsection{Conditional Distribution/PMF (Discrete)}
For discrete random variables $X,Y$ and $P(Y=y) > 0$,
\[P_{X\mid Y}(x\mid y) = P(X=x\mid Y=y) = \frac{P(X=x, Y=y)}{P(Y=y)}\]
The probability of $X=x$ given $Y=y$.

\medskip

\noindent Multiplicative Rule:
\[P(X=x, Y=y) = P(X=x\mid Y=y)\cdot P(Y=y)\]
The Joint Probability is the product of the Marginal and the Conditional.

\subsubsection{Example (Sampling Without Replacement)}

Box contains 3 white and 3 black balls. Draw 2 balls without replacement. \\
$X = $ result of first draw ($1 = $ white, $0 = $ black) \\
$Y = $ result of second draw ($1 = $ white, $0 = $ black)

\noindent PMF:
\begin{align*}
    P(X = 0, Y = 0) &= \frac{3}{6} \times \frac{2}{5} = \frac{1}{5}, & P(X = 1, Y = 1) &= \frac{1}{5} \\
    P(X = 0, Y = 1) &= \frac{3}{6} \times \frac{3}{5} = \frac{3}{10}, & P(X = 1, Y = 0) &= \frac{3}{10}
\end{align*}

\noindent Marginals:
\begin{equation*}
    P(X = 1) = \frac{1}{2}, \quad P(Y = 1) = \frac{1}{2}
\end{equation*}

\noindent We can conclude $X$ and $Y$ are not independent.

\bigskip

\noindent Conditionals:

\noindent Since $P(X = 1, Y = 1) = \frac{1}{5}$ and $P(Y = 1) = \frac{1}{2}$, the conditional distribution is:
\begin{equation*}
    P(X = 1 \mid Y = 1) = \frac{2}{5} < \frac{1}{2} = P(X=1)
\end{equation*}

\noindent Knowing the second ball is white makes the first less likely to be white.

\subsubsection{Joint Distribution/PDF (Continuous)}
For continuous random variables $X,Y$ and $A\subset \mathbb{R}^2$,
\[P((X,Y)\in A) = \iint_A f_{X,Y}(x,y)\:\mathrm{d}x\:\mathrm{d}y\]
where $f_{X,Y}(x,y)$ is the joint PDF.

\medskip

\noindent Properties:
\begin{enumerate}
    \item $f_{X,Y}(x,y)\geq 0$
    \item $\iint_{\mathbb{R}^2}f_{X,Y}(x,y)\:\mathrm{d}x\:\mathrm{d}y = 1$
\end{enumerate}

\subsubsection{Marginal Distribution/PDF (Continuous)}
Given the joint PDF $f_{X,Y}(x,y)$, the marginal distributions/PDFs of $X$ and $Y$
are found by integrating out the other variable.
\[f_X(x) = \int_{-\infty}^{+\infty}f_{X,Y}(x,y)\:\mathrm{d}y\]
\[f_Y(y) = \int_{-\infty}^{+\infty}f_{X,Y}(x,y)\:\mathrm{d}x\]

\noindent Properties:
\begin{enumerate}
    \item $f_X(x) \geq 0$
    \item $\int_{-\infty}^{+\infty}f_X(x)\:\mathrm{d}x = 1$
\end{enumerate}

\subsubsection{Conditional Distribution/PDF (Continuous)}
For continuous random variables $X,Y$ and $f_Y(y) > 0$,
\[f_{X\mid Y}(x\mid y) = \frac{f_{X,Y}(x,y)}{f_Y(y)}\]

\bigskip

\noindent Multiplicative Rule: $f_{X,Y}(x,y) = f_{X\mid Y}(x\mid y)f_Y(y)$

\subsection{Independence of Random Variables}
Two random variables $X,Y$ are independent if, for event sets $A,B$,
\[P(X\in A,Y\in B) = P(X\in A)\cdot P(Y\in B) \iff P_{X,Y}(A \times B) = P_X(A)\cdot P_Y(B)\]

\noindent Discrete Case: $A\perp B \iff p_{X,Y}{x,y} = p_X(x)p_Y(y)$

\noindent Continuous Case: $A\perp B \iff f_{X,Y}{x,y} = f_X(x)f_Y(y)$

\medskip

\noindent Conditional PMF and PDF: $p_{X\mid Y}(x\mid y) = p_X(x)$ and $f_{X,Y}(x,y) = f_X(x)$
